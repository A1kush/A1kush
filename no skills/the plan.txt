```js
function notify(msg, color){
  const wrap = notifWrap();
  const d = document.createElement('div');
  d.className = 'toast';

  d.textContent = msg;
```

## 3) Fix the `domSprites` null crash (clone bar parent)

**Why:** Your error (“Cannot read properties of null (reading 'appendChild')”) happens when code tries to append to `#domSprites` before it exists. Sometimes the element isn’t in the document when a hook fires.

**Safe parent helper:**

```js
function getDomSprites(){
  let d = document.getElementById('domSprites');
  if(!d){
    d = document.createElement('div');
    d.id = 'domSprites';
    d.style.cssText = 'position:absolute;left:0;top:0;right:0;bottom:0;pointer-events:none';
    document.getElementById('wrap')?.appendChild(d) || document.body.appendChild(d);
  }
  return d;
}
// When adding clone bars or DOM sprites:
getDomSprites().appendChild(someChild);
```

Note: `#domSprites` is already in your file near the core script; this guard simply prevents failures on timing or future refactors .

## 5) Give mobs proper projectiles (shooters and occasional melee throw)

**Why:** You asked for mobs to shoot. You already have `st.eShots` and a shooter role; this wires consistent bullet creation.

**Bullet helper + usage:**

```js
function fireEnemyBullet(e, speed=320, dmg=14){
  const L = leader(); if(!L) return;
  const dx = (L.x - e.x), dy = (L.y - e.y);
  const mag = Math.hypot(dx, dy) || 1;
  st.eShots.push({
    x: e.x - 18, y: e.y - 30,
    vx: speed * (dx/mag), vy: speed * (dy/mag),
    dmg, life: 2200, w: 24, h: 24, color:'#ffbb66'
  });
  e._justFired = true;
}

function tickEnemyShooting(e, dt){
  e.fireCD = (e.fireCD ?? (800 + Math.random()*600)) - dt*1000;
  if(e.fireCD <= 0){
    const dmg = e.kind === 'boss' ? 26 : 12;
    fireEnemyBullet(e, 360, dmg);
    e.fireCD = (e.kind === 'boss' ? 600 : 1000) + Math.random()*400;
  }
}

// in your enemy update:
if (e.role === 'shooter') tickEnemyShooting(e, dt);
// melee gets a rare throw:
if (e.role === 'melee' && Math.random() < 0.003) fireEnemyBullet(e, 300, 10);
```

## 6) DebugSkills v2.3 (probe×3 → Secret (5s grace) → X-slash WaveKill)

**Why:** Auto-unstick flow per your spec, with timers per wave and “stuck” detection when enemy HP pool is flat ≥1.5s.

**Drop-in module:**

```js
const dbg = {
  armed: true, waveStart: 0, step: 0,
  tProbe: 0, probesFired: 0, lastHP: 0, stableT: 0,
  unlockMin: 10000, unlockMax: 20000, hardcapMin: 30000, hardcapMax: 40000
};
function totalEnemyHP(){ return st.enemies.reduce((s,e)=>s+Math.max(0,e.hp),0); }
function dbgWaveStart(){
  dbg.waveStart = performance.now();
  dbg.step = 0; dbg.tProbe = 0; dbg.probesFired = 0; dbg.lastHP = totalEnemyHP(); dbg.stableT = 0;
}
function dbgTick(dt){
  if(!dbg.armed) return;

  // Show allies & their projectiles while armed
  st.players.forEach(p=>p._forceVisible=true);

  // Stable HP detector (≥1.5s)
  const hp = totalEnemyHP();
  if (Math.abs(hp - dbg.lastHP) <= 0.01) { dbg.stableT += dt*1000; } else { dbg.stableT = 0; }
  dbg.lastHP = hp;

  const t = performance.now() - dbg.waveStart;
  const unlock = dbg._unlock ??= (dbg.unlockMin + Math.random()*(dbg.unlockMax-dbg.unlockMin));
  const hardcap = dbg._hardcap ??= (dbg.hardcapMin + Math.random()*(dbg.hardcapMax-dbg.hardcapMin));

  // Step 0: after unlock window or if stuck → fire up to 3 probes (30 dmg homing)
  if (dbg.step === 0 && (t >= unlock || dbg.stableT >= 1500)) {
    if (dbg.probesFired < 3) {
      fireHomingProbe(30); dbg.probesFired++; dbg.tProbe = performance.now();
      if (dbg.probesFired >= 3) dbg.step = 1;
    } else dbg.step = 1;
  }
  // Step 1: 5s grace Secret (A1 X1 costless) if still stuck
  if (dbg.step === 1 && (dbg.stableT >= 1500 || (performance.now()-dbg.tProbe) >= 10000)) {
    castA1SecretFree();
    dbg.step = 2;
    dbg._secretT = performance.now();
  }
  // Step 2: at hardcap or still stuck after Secret → red X VFX + WaveKill → nextWave
  if (dbg.step <= 2 && (t >= hardcap || dbg.stableT >= 1500 || (performance.now()-dbg._secretT) >= 5000)) {
    waveKillVFX();
    st.enemies.slice().forEach(e=>{ e.hp = 0; onEnemyKilled(e); });
    dbg.step = 3;
  }
}
function fireHomingProbe(dmg){
  const L = leader(); if(!L) return;
  const e = st.enemies.find(x=>x.hp>0); if(!e) return;
  const dx = e.x - L.x, dy = e.y - L.y, mag = Math.hypot(dx,dy)||1;
  st.shots.push({ x:L.x+20,y:L.y-40, vx:420*(dx/mag), vy:420*(dy/mag), dmg, life:1600, homing:true, ownerId:'ally', color:'#ffaa00' });
  notify('Debug Probe', '#ffaa00');
}
function castA1SecretFree(){
  const a = st.players.find(p=>p.id==='A1' && !p.isDefeated); if(!a) return;
  a.secretGauge = a.secretMax; a.hp = Math.max(1, a.hp); // ensure castable without cost
  useSkillWrapper(a, 'X1');
  notify('Debug Secret', '#ffaa00');
}
function waveKillVFX(){
  st.effects.push({kind:'impact',x:DESIGN_W/2,y:DESIGN_H/2,color:'#ff3366',life:1000,max:1000});
}
```
